
test('⑨ ★role3-pilot 的严格口径（drops:[] 是如实，非缺失）', () => {
  // 背景（2026-09-22）：role3-pilot 的口径**比 drops 数组更严**——无效项（解析失败/契约违约/调用失败）
  //   不丢弃，而是留在 rows 里以 `_valid:false` 标记（resolve=null ⇒ gate 判 no_anchor）
  //   ⇒ 分母天然＝提议全集。本批补显式 `drops: []` ＋ 说明，让 gate 能报**严格口径**。
  const src = fs.readFileSync(SCRIPT, 'utf8');
  assert.ok(/drops: \[\]/.test(src), '★须显式输出 drops: []（否则 gate 只报候选口径）');
  assert.ok(/drops_note/.test(src), '须附说明（防后人误以为「空数组＝没做留痕」）');
  assert.ok(/_valid:false/.test(src) || /_valid/.test(src), '须保留 _valid 标记机制');
  // ★口径说明须点明「比 drops 更严」
  assert.ok(/不丢弃任何提议|比 drops 口径更严|分母天然/.test(src), '★说明须点明该口径比 drops 更严');
  // 产物锁：若 rows 件存在且已含 drops，须为空数组
  const rp = path.join(ROOT, 'p1b/sim/out/role3-pilot-20260919.rows.json');
  if (fs.existsSync(rp)) {
    const j = JSON.parse(fs.readFileSync(rp, 'utf8'));
    if ('drops' in j) assert.deepEqual(j.drops, [], 'drops 须为空数组（该 runner 不丢弃提议）');
  }
});
